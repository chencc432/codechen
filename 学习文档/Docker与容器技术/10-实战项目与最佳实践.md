### 1.3 Backend Dockerfile

```dockerfile
FROM python:3.11-slim AS builder

WORKDIR /app

RUN apt-get update && \
    apt-get install -y --no-install-recommends gcc libpq-dev && \
    rm -rf /var/lib/apt/lists/*

COPY requirements.txt .
RUN pip install --no-cache-dir --user -r requirements.txt

FROM python:3.11-slim

RUN apt-get update && \
    apt-get install -y --no-install-recommends libpq5 curl && \
    rm -rf /var/lib/apt/lists/* && \
    groupadd -r appuser && useradd -r -g appuser -s /sbin/nologin appuser

WORKDIR /app

COPY --from=builder /root/.local /root/.local
ENV PATH=/root/.local/bin:$PATH

COPY . .

RUN chown -R appuser:appuser /app
USER appuser

EXPOSE 8000

HEALTHCHECK --interval=30s --timeout=5s --retries=3 \
    CMD curl -f http://localhost:8000/health || exit 1

CMD ["gunicorn", "--bind", "0.0.0.0:8000", "--workers", "4", \
     "--access-logfile", "-", "app.main:app"]
```

### 1.5 Nginx 反向代理配置

```nginx
# nginx/nginx.conf
upstream backend {
    server backend:8000;
}

server {
    listen 80;
    server_name localhost;

    # 前端静态文件
    location / {
        root /usr/share/nginx/html;
        try_files $uri $uri/ /index.html;
    }

    # API 反向代理
    location /api/ {
        proxy_pass http://backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # 健康检查
    location /health {
        access_log off;
        return 200 "healthy\n";
    }
}
```

### 1.6 docker-compose.yml

```yaml
services:
  nginx:
    build: ./nginx
    ports:
      - "${APP_PORT:-80}:80"
    depends_on:
      frontend:
        condition: service_healthy
      backend:
        condition: service_healthy
    restart: unless-stopped
    networks:
      - frontend-net
      - backend-net

  frontend:
    build: ./frontend
    expose:
      - "80"
    healthcheck:
      test: ["CMD", "wget", "-qO-", "http://localhost:80/"]
      interval: 30s
      timeout: 5s
      retries: 3
    restart: unless-stopped
    networks:
      - frontend-net

  backend:
    build: ./backend
    expose:
      - "8000"
    environment:
      - DATABASE_URL=postgresql://${DB_USER}:${DB_PASS}@db:5432/${DB_NAME}
      - REDIS_URL=redis://cache:6379/0
      - SECRET_KEY=${SECRET_KEY}
    depends_on:
      db:
        condition: service_healthy
      cache:
        condition: service_started
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/health"]
      interval: 30s
      timeout: 5s
      retries: 3
    restart: unless-stopped
    networks:
      - backend-net

  db:
    image: postgres:16-alpine
    environment:
      POSTGRES_USER: ${DB_USER}
      POSTGRES_PASSWORD: ${DB_PASS}
      POSTGRES_DB: ${DB_NAME}
    volumes:
      - pg-data:/var/lib/postgresql/data
      - ./scripts/init-db.sql:/docker-entrypoint-initdb.d/init.sql
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${DB_USER} -d ${DB_NAME}"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped
    networks:
      - backend-net

  cache:
    image: redis:7-alpine
    command: redis-server --appendonly yes --maxmemory 256mb --maxmemory-policy allkeys-lru
    volumes:
      - redis-data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped
    networks:
      - backend-net

volumes:
  pg-data:
  redis-data:

networks:
  frontend-net:
  backend-net:
```

## 二、Docker 安全最佳实践

### 2.1 镜像安全

```dockerfile
# 1. 使用官方基础镜像 + 固定版本
FROM python:3.11.7-slim    # ✅
# FROM python:latest        # ❌

# 2. 不以 root 用户运行
RUN groupadd -r app && useradd -r -g app app
USER app

# 3. 使用多阶段构建，最终镜像不包含构建工具
FROM golang:1.22 AS builder
# ... 构建 ...
FROM scratch
COPY --from=builder /app/binary /app/binary

# 4. 最小化安装包
RUN apt-get install -y --no-install-recommends curl && \
    rm -rf /var/lib/apt/lists/*
```

### 2.2 运行时安全

```bash
# 1. 限制资源，防止容器吃光系统资源
docker run --cpus=2 --memory=512m --pids-limit=100 my-app

# 2. 只读文件系统
docker run --read-only --tmpfs /tmp my-app

# 3. 不使用特权模式
# ❌ docker run --privileged my-app
# ✅ 只赋予需要的 capability
docker run --cap-drop=ALL --cap-add=NET_BIND_SERVICE my-app

# 4. 限制网络
docker run --network=none my-batch-job

# 5. 不暴露不必要的端口
# ❌ docker run -p 0.0.0.0:3306:3306 mysql
# ✅ docker run -p 127.0.0.1:3306:3306 mysql
```

### 2.3 密钥管理

```bash
# ❌ 错误：把密钥写在 Dockerfile 或镜像中
ENV API_KEY=super-secret-key

# ✅ 正确：运行时传入
docker run -e API_KEY=super-secret-key my-app

# ✅ 更好：使用 env-file
docker run --env-file .env.secret my-app

# ✅ 最好：使用 Docker Secrets（Swarm 模式）或外部密钥管理
```

### 2.4 镜像扫描

```bash
# 使用 Docker Scout 扫描漏洞
docker scout cves my-app:latest

# 使用 Trivy（推荐）
docker run --rm -v /var/run/docker.sock:/var/run/docker.sock \
    aquasec/trivy image my-app:latest
```

## 三、Docker 性能优化

### 3.1 镜像体积优化

```
优化前                        优化后
python:3.11     (1.0 GB)  →  python:3.11-slim  (150 MB)
                          →  python:3.11-alpine ( 50 MB)
                          →  多阶段 + scratch   (  5 MB)  # 仅限 Go 等
```

**优化检查清单**：
- [ ] 使用 slim 或 alpine 基础镜像
- [ ] 多阶段构建，构建工具不进最终镜像
- [ ] 合并 RUN 命令，同层清理
- [ ] .dockerignore 排除不需要的文件
- [ ] 不安装不必要的包（`--no-install-recommends`）
- [ ] 清理包管理器缓存（`rm -rf /var/lib/apt/lists/*`）

### 3.2 构建速度优化

```dockerfile
# 1. 先复制依赖文件，再复制代码（利用缓存）
COPY requirements.txt .
RUN pip install -r requirements.txt
COPY . .

# 2. 使用 BuildKit 缓存挂载
RUN --mount=type=cache,target=/root/.cache/pip \
    pip install -r requirements.txt

# 3. 并行构建（BuildKit 自动优化）
```

### 3.3 运行时性能

```bash
# 1. 合理的资源分配
docker run --cpus=4 --memory=4g my-app

# 2. 使用 host 网络（减少 NAT 开销）
docker run --network host my-app

# 3. 合理的日志驱动
docker run --log-driver json-file --log-opt max-size=10m --log-opt max-file=3 my-app

# 4. 使用 tmpfs 提升临时文件读写性能
docker run --tmpfs /tmp:size=256m my-app
```

## 四、日常运维速查

### 4.1 日志管理

```bash
# 查看容器日志大小
docker inspect --format='{{.LogPath}}' my-container
ls -lh $(docker inspect --format='{{.LogPath}}' my-container)

# 配置全局日志限制（/etc/docker/daemon.json）
{
    "log-driver": "json-file",
    "log-opts": {
        "max-size": "10m",
        "max-file": "3"
    }
}
```

### 4.2 磁盘管理

```bash
# 查看 Docker 磁盘使用
docker system df
docker system df -v

# 清理所有未使用资源
docker system prune          # 安全清理
docker system prune -a       # 激进清理（包括所有未使用镜像）
docker system prune --volumes  # 连 Volume 也清（数据会丢！）

# 定时清理（加到 crontab）
# 每天凌晨 3 点清理 7 天前的资源
0 3 * * * docker system prune -f --filter "until=168h"
```

### 4.3 监控

```bash
# 实时资源监控
docker stats

# 格式化输出
docker stats --format "table {{.Name}}\t{{.CPUPerc}}\t{{.MemUsage}}\t{{.NetIO}}"

# 查看容器事件流
docker events
docker events --filter 'type=container' --filter 'event=die'
```

### 4.4 故障排查流程

```
容器出问题了？按这个流程排查：

1. 看状态
   docker ps -a
   → 容器是 running 还是 exited？exit code 是多少？

2. 看日志
   docker logs --tail 100 my-container
   → 有没有报错信息？

3. 看资源
   docker stats my-container
   → CPU 或内存是不是打满了？

4. 进容器排查
   docker exec -it my-container sh
   → 看看进程、网络、文件是否正常

5. 看变更
   docker diff my-container
   → 哪些文件被修改了？

6. 看详情
   docker inspect my-container
   → 检查配置、网络、挂载是否正确

7. 网络排查
   docker exec my-container ping target
   docker exec my-container curl http://target:port
   → 网络通不通？DNS 解析对不对？
```

## 五、Docker + nerdctl 终极命令速查表

### 镜像操作

| 操作 | docker / nerdctl |
|------|-----------------|
| 拉取镜像 | `pull nginx:1.25` |
| 查看镜像 | `images` |
| 构建镜像 | `build -t app:v1 .` |
| 删除镜像 | `rmi app:v1` |
| 打标签 | `tag app:v1 registry/app:v1` |
| 推送 | `push registry/app:v1` |
| 导出 | `save -o app.tar app:v1` |
| 导入 | `load -i app.tar` |
| 查看历史 | `history app:v1` |
| 清理 | `image prune -a` |

### 容器操作

| 操作 | docker / nerdctl |
|------|-----------------|
| 运行 | `run -d --name web -p 80:80 nginx` |
| 查看 | `ps -a` |
| 停止 | `stop web` |
| 启动 | `start web` |
| 重启 | `restart web` |
| 删除 | `rm -f web` |
| 进入 | `exec -it web bash` |
| 日志 | `logs -f --tail 100 web` |
| 资源 | `stats` |
| 复制 | `cp web:/file ./local` |
| 详情 | `inspect web` |
| 进程 | `top web` |

### 网络操作

| 操作 | docker / nerdctl |
|------|-----------------|
| 创建 | `network create my-net` |
| 查看 | `network ls` |
| 详情 | `network inspect my-net` |
| 连接 | `network connect my-net web` |
| 断开 | `network disconnect my-net web` |
| 删除 | `network rm my-net` |

### Volume 操作

| 操作 | docker / nerdctl |
|------|-----------------|
| 创建 | `volume create my-data` |
| 查看 | `volume ls` |
| 详情 | `volume inspect my-data` |
| 删除 | `volume rm my-data` |
| 清理 | `volume prune` |

### Compose 操作

| 操作 | docker compose / nerdctl compose |
|------|--------------------------------|
| 启动 | `compose up -d` |
| 构建并启动 | `compose up -d --build` |
| 停止 | `compose stop` |
| 停止清理 | `compose down` |
| 状态 | `compose ps` |
| 日志 | `compose logs -f` |
| 进入 | `compose exec web bash` |
| 扩缩 | `compose up -d --scale worker=3` |

### 系统操作

| 操作 | docker / nerdctl |
|------|-----------------|
| 版本 | `version` |
| 信息 | `info` 或 `system info` |

## 六、进阶学习路线

```

  ✅ Docker 基础 → 镜像、容器、网络、存储、Compose

  └── 把 Docker 集成到自动化流水线
```
